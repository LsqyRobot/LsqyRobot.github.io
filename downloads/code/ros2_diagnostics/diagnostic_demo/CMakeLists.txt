cmake_minimum_required(VERSION 3.8)
project(diagnostic_demo)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(diagnostic_node src/diagnostic_node.cpp)
target_compile_features(diagnostic_node PUBLIC cxx_std_17)
ament_target_dependencies(diagnostic_node rclcpp std_msgs)
install(TARGETS diagnostic_node DESTINATION lib/${PROJECT_NAME})
ament_package()
